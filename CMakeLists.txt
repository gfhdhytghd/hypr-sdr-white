cmake_minimum_required(VERSION 3.19)
project(hypr-sdr-white VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPR REQUIRED IMPORTED_TARGET hyprland)
add_library(hypr-sdr-white SHARED src/main.cpp)
target_link_libraries(hypr-sdr-white PRIVATE PkgConfig::HYPR)
target_compile_options(hypr-sdr-white PRIVATE -Wall -Wextra -Wno-unused-parameter)
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag(-fno-gnu-unique HAS_NO_GNU_UNIQUE)
if(HAS_NO_GNU_UNIQUE)
    target_compile_options(hypr-sdr-white PRIVATE -fno-gnu-unique)
endif()
install(TARGETS hypr-sdr-white LIBRARY DESTINATION lib)
